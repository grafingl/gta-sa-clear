{$CLEO .cs}
{$USE CLEO+}
{$USE newOpcodes}
{$USE ini}

0000: NOP

script_name "StartBP"

// By ArtemQa146


repeat
wait 500
0AF0: 1@ = read_int_from_ini_file "cleo\PedsModels.ini" section "Models" key "Loaded1"
until 1@ == 1
1@ = 0
0AF1: write_int 1@ to_ini_file "cleo\PedsModels.ini" section "Models" key "Loaded1"

timera = 0
timerb = 0

const
    pause = 300
    umb1 = 5@
    umb2 = 6@
    umb3 = 7@
end


0AF0: umb1 = read_int_from_ini_file "cleo\PedsModels.ini" section "Models" key "umb1"
0AF0: umb2 = read_int_from_ini_file "cleo\PedsModels.ini" section "Models" key "umb2"
0AF0: umb3 = read_int_from_ini_file "cleo\PedsModels.ini" section "Models" key "umb3"

while true
wait pause
    if
        player.Defined(0)
    then
        06AC: 1@ = actor $PLAYER_ACTOR movement_speed
        if
            1@ > 50.0
        then
            continue
        end


            0AAA: 1@ = get_script_struct_named "WkPeds"
            if
                1@ == 0
            then
                0842: 2@ = player $PLAYER_CHAR town_number
                0D59: get_current_weather_to 1@
                if and
                    not 1@ == 8 // not weather rain
                    not 1@ == 16 // not weather rain 2
                    not 2@ == 0
                    $ACTIVE_INTERIOR == 0
                then
                    0A92: stream_custom_script "PedsWalk.qa"
                    wait 2000
                end
            end
            
            wait pause
            
            0AAA: 1@ = get_script_struct_named "CsPeds"
            if
                1@ == 0
            then
                0842: 2@ = player $PLAYER_CHAR town_number
                0D59: get_current_weather_to 1@
                if and
                    not 1@ == 8 // not weather rain
                    not 1@ == 16 // not weather rain 2
                    2@ == 0
                    $ACTIVE_INTERIOR == 0
                then
                    0A92: stream_custom_script "PedsCountrySIde.qa"
                    wait 2000
                end
            end
            
            wait pause
            
            0AAA: 1@ = get_script_struct_named "Wk2Peds"
            if
                1@ == 0
            then
                0842: 2@ = player $PLAYER_CHAR town_number
                0D59: get_current_weather_to 1@
                if and
                    not 1@ == 8 // not weather rain
                    not 1@ == 16 // not weather rain 2
                    $ACTIVE_INTERIOR == 0
                    $ONMISSION == 0
                then
                    0A92: stream_custom_script "PedsWalk2.qa"
                    wait 2000
                end
            end
            
            wait pause
            
            0AAA: 1@ = get_script_struct_named "RnPeds"
            if
                1@ == 0
            then
                0D59: get_current_weather_to 1@
                if or
                    1@ == 8 // not weather rain
                    1@ == 16 // not weather rain 2
                then
                    if
                        $ACTIVE_INTERIOR == 0
                    then
                        0A92: stream_custom_script "PedsRain.qa" umb1 umb2 umb3
                        wait 2000
                    else
                        wait 500
                    end
                end
            end
            
            wait pause
            
            0AAA: 1@ = get_script_struct_named "WrkPeds"
            if
                1@ == 0
            then
                if
                    $ACTIVE_INTERIOR == 0
                then
                    0E40: get_current_hour 2@
                    0D59: get_current_weather_to 1@
                    if and
                        2@ >= 7
                        2@ < 18
                        not 1@ == 8 // not weather rain
                        not 1@ == 16 // not weather rain 2
                    then
                        if or
                            0154:   actor $PLAYER_ACTOR in_zone 'LDOC'
                            0154:   actor $PLAYER_ACTOR in_zone 'LAIR'
                            0154:   actor $PLAYER_ACTOR in_zone 'SFAIR'
                            0154:   actor $PLAYER_ACTOR in_zone 'EASB'
                            0154:   actor $PLAYER_ACTOR in_zone 'VAIR'
                            0154:   actor $PLAYER_ACTOR in_zone 'KACC'
                        then
                            0A92: stream_custom_script "PedsWork.qa"
                            wait 2000
                        end
                    end
                end
            end
            
            wait pause
            
            0AAA: 1@ = get_script_struct_named "BchPeds"
            if
                1@ == 0
            then
                if
                    $ACTIVE_INTERIOR == 0
                then
                    0E40: get_current_hour 2@
                    0D59: get_current_weather_to 1@
                    if and
                        2@ >= 5
                        2@ < 23
                        not 1@ == 8 // not weather rain
                        not 1@ == 16 // not weather rain 2
                    then
                        if or
                            0154:   actor $PLAYER_ACTOR in_zone 'SMB'
                            0154:   actor $PLAYER_ACTOR in_zone 'VERO'
                            0154:   actor $PLAYER_ACTOR in_zone 'PLS'
                            0154:   actor $PLAYER_ACTOR in_zone 'OCEAF'
                        then
                            0A92: stream_custom_script "PedsBeach.qa"
                            wait 2000
                        end
                    end
                end
            end
            
            wait pause
            
            0AAA: 1@ = get_script_struct_named "HmPeds"
            if
                1@ == 0
            then
                if and
                    timera > 10000
                    $ACTIVE_INTERIOR == 0
                    $ONMISSION == 0
                then
                    timera = 0
                    0A92: stream_custom_script "PedsHomeless.qa"
                    wait 2000
                end
            else
                timera = 0
            end
            
            wait pause
            
            0AAA: 1@ = get_script_struct_named "StPeds"
            if
                1@ == 0
            then
                if and
                    timerb > 5000
                    $ACTIVE_INTERIOR == 0
                then
                    timerb = 0
                    0A92: stream_custom_script "PedsSeat.qa"
                    wait 2000
                end
            else
                timerb = 0
            end
            
            
    end
end
